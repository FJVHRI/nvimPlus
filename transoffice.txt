有！**ECDICT**（开源英汉词典数据库）完全符合你的需求，包含音标、词性、例句、词形变化（同根词），且完全离线。

## 方案：ECDICT + Neovim 原生查询（最快最稳）

### 1. 下载离线词典数据库

```bash
mkdir -p ~/.local/share/nvim/dict
cd ~/.local/share/nvim/dict

# 下载 ECDICT SQLite 版（约 150MB，包含 50万+ 词条）
wget https://github.com/skywind3000/ECDICT/releases/download/1.0.28/ecdict-sqlite-28.zip

# 解压
unzip ecdict-sqlite-28.zip
# 得到 stardict.db（这就是你要的离线词典）

# 验证大小（正常约 150MB）
ls -lh stardict.db
```

### 2. Neovim 配置（替换 trans.lua）

```lua
-- ~/.config/nvim/lua/plugins/trans.lua
return {
  "JuanZoran/Trans.nvim",
  enabled = false, -- 禁用有问题的插件
}

-- 自定义 ECDICT 离线词典查询
local dict_path = vim.fn.stdpath("data") .. "/dict/stardict.db"

local function query_ecdict(word)
  if vim.fn.filereadable(dict_path) == 0 then
    return nil, "词典数据库未找到，请下载: " .. dict_path
  end
  
  -- 使用 sqlite3 命令行查询
  local cmd = string.format(
    'sqlite3 %s "SELECT phonetic, translation, exchange, detail FROM stardict WHERE word = \'%s\' LIMIT 1"',
    dict_path,
    word:gsub("'", "''") -- 防 SQL 注入
  )
  
  local output = vim.fn.system(cmd)
  if vim.v.shell_error ~= 0 or output == "" then
    return nil, "未找到单词: " .. word
  end
  
  -- 解析输出 (phonetic|translation|exchange|detail)
  local parts = vim.split(output:gsub("\n$", ""), "|", { plain = true })
  return {
    phonetic = parts[1] or "",
    translation = parts[2] or "",
    exchange = parts[3] or "", -- 时态/复数变化
    detail = parts[4] or "",   -- JSON 格式的详细释义和例句
  }
end

local function show_result(word, data)
  local lines = {}
  local highlights = {}
  
  -- 标题
  table.insert(lines, "╭─────────────────────────────────────╮")
  table.insert(lines, string.format("│  %-33s  │", word))
  table.insert(lines, "├─────────────────────────────────────┤")
  
  -- 音标
  if data.phonetic and data.phonetic ~= "" then
    table.insert(lines, string.format("│  音标: %s", data.phonetic))
    table.insert(highlights, { line = #lines - 1, col = 2, len = 4, group = "Comment" })
  end
  
  -- 词性+释义（translation 字段已包含词性，如 "n. 计算机..."）
  if data.translation and data.translation ~= "" then
    for line in data.translation:gmatch("[^\\n]+") do
      table.insert(lines, string.format("│  %s", line))
    end
  end
  
  -- 词形变化（同根词/变形）
  if data.exchange and data.exchange ~= "" then
    table.insert(lines, "├─────────────────────────────────────┤")
    table.insert(lines, "│  【词形变化】")
    -- 解析 exchange 格式: "d:computed/ing:computing/s:computers"
    for part in data.exchange:gmatch("[^/]+") do
      local key, val = part:match("(.+):(.+)")
      if key and val then
        local label = ({d="过去式",p="过去分词",i="现在分词",s="复数",r="比较级",t="最高级",3="第三人称单数"})[key] or key
        table.insert(lines, string.format("│    %s: %s", label, val))
      end
    end
  end
  
  -- 详细释义和例句（解析 JSON detail 字段）
  if data.detail and data.detail ~= "" then
    local ok, detail_json = pcall(vim.json.decode, data.detail)
    if ok and type(detail_json) == "table" then
      table.insert(lines, "├─────────────────────────────────────┤")
      table.insert(lines, "│  【例句】")
      
      -- 遍历词性分类（noun, verb 等）
      for pos, defs in pairs(detail_json) do
        if type(defs) == "table" then
          for _, def in ipairs(defs) do
            if def.definition then
              table.insert(lines, string.format("│  %s: %s", pos:upper(), def.definition))
            end
            if def.example then
              table.insert(lines, string.format("│    ➜ %s", def.example))
              if def.translation then
                table.insert(lines, string.format("│      %s", def.translation))
              end
            end
            -- 限制例句数量，避免窗口过高
            if #lines > 30 then break end
          end
        end
      end
    end
  end
  
  table.insert(lines, "╰─────────────────────────────────────╯")
  
  -- 创建浮动窗口
  local buf = vim.api.nvim_create_buf(false, true)
  vim.api.nvim_buf_set_lines(buf, 0, -1, false, lines)
  vim.api.nvim_buf_set_option(buf, "modifiable", false)
  vim.api.nvim_buf_set_option(buf, "filetype", "dictionary")
  
  local width = math.min(60, vim.o.columns - 4)
  local height = math.min(#lines, vim.o.lines - 4)
  
  local win = vim.api.nvim_open_win(buf, true, {
    relative = "editor",
    width = width,
    height = height,
    col = (vim.o.columns - width) / 2,
    row = (vim.o.lines - height) / 2,
    style = "minimal",
    border = "rounded",
    title = " 离线词典 (ECDICT) ",
    title_pos = "center",
  })
  
  -- 高亮
  vim.api.nvim_win_set_option(win, "winhl", "Normal:Normal,FloatBorder:FloatBorder")
  -- 添加简单高亮（可选）
  local ns = vim.api.nvim_create_namespace("ecdict")
  for _, hl in ipairs(highlights) do
    vim.api.nvim_buf_add_highlight(buf, ns, hl.group, hl.line, hl.col, hl.col + hl.len)
  end
  
  -- 按 q 关闭
  vim.api.nvim_buf_set_keymap(buf, "n", "q", "<cmd>close<cr>", { noremap = true, silent = true })
  vim.api.nvim_buf_set_keymap(buf, "n", "<Esc>", "<cmd>close<cr>", { noremap = true, silent = true })
end

-- 主函数
local function translate(mode)
  local word
  if mode == "v" then
    -- 获取选中文本
    local old_reg = vim.fn.getreg('"')
    vim.cmd('normal! ""y')
    word = vim.fn.getreg('"'):gsub("^%s*(.-)%s*$", "%1") -- trim
    vim.fn.setreg('"', old_reg)
  else
    word = vim.fn.expand("<cword>")
  end
  
  if not word or word == "" then return end
  
  vim.notify("查询中: " .. word, vim.log.levels.INFO, { timeout = 500 })
  
  -- 异步查询避免卡顿
  vim.defer_fn(function()
    local data, err = query_ecdict(word:lower())
    if not data then
      vim.notify(err or "查询失败", vim.log.levels.WARN)
      return
    end
    
    vim.schedule(function()
      show_result(word, data)
    end)
  end, 10)
end

-- 快捷键
vim.keymap.set("n", "mm", function() translate("n") end, { desc = "离线词典查单词" })
vim.keymap.set("v", "mm", function() translate("v") end, { desc = "离线词典查选中文本" })
vim.keymap.set("n", "mi", function()
  vim.ui.input({ prompt = "输入单词: " }, function(input)
    if input then translate("n", input) end
  end)
end, { desc = "离线词典输入查询" })
```

### 4. 数据库特点

- **音标**：KK/IPA 音标（如 `/kəm'pju:tə(r)/`）
- **词性**：`n.` `v.` `adj.` 等前缀已包含在释义中
- **用法/搭配**：部分词条包含 `detail` 字段的用法说明
- **例句**：包含双语例句（英文+中文翻译）
- **同根词/变形**：`exchange` 字段包含时态、复数、分词等变化形式

### 5. 与 GoldenDict 对比

| 特性 | ECDICT + Nvim | GoldenDict |
|------|---------------|------------|
| **速度** | ⚡ 毫秒级（SQLite） | 较快 |
| **例句质量** | 基础例句 | 朗文/牛津例句更地道 |
| **词根词缀** | 词形变化 | 可能更详细（取决于词典文件）|
| **内存占用** | 仅 SQLite 文件 (~150MB) | 需运行 GoldenDict 程序 |
| **界面** | Neovim 浮动窗口 | 独立 GUI 窗口 |

**建议**：先试用 ECDICT，如果例句质量不满足，再下载 **朗文/牛津 MDX 词典** 用 GoldenDict 查看。

现在执行第一步下载命令，然后重启 nvim 按 `mm` 即可离线查词，**无需 AI 模型，无需显存**。
