-- ~/.config/nvim/lua/plugins/trans.lua

return {
  {
    "JuanZoran/Trans.nvim",
    enabled = false,
  },

  {
    name = "ecdict-offline",
    dir = vim.fn.stdpath("config"),
    lazy = false,
    config = function()
      local dict_path = vim.fn.stdpath("config") .. "/dict/stardict.db"
      local audio_cache_dir = vim.fn.stdpath("cache") .. "/dict_audio"
      vim.fn.mkdir(audio_cache_dir, "p")

      -- 查询本地 ECDICT
      local function query_ecdict(word)
        if vim.fn.filereadable(dict_path) == 0 then
          return nil, "词典未找到: " .. dict_path
        end
        
        local cmd = string.format(
          'sqlite3 "%s" "SELECT phonetic, translation, exchange, detail FROM stardict WHERE word = \'%s\' LIMIT 1"',
          dict_path,
          word:gsub("'", "''")
        )
        
        local output = vim.fn.system(cmd)
        if vim.v.shell_error ~= 0 or output == "" then
          return nil, "未找到: " .. word
        end
        
        local parts = vim.split(output:gsub("\n$", ""), "|", { plain = true })
        return {
          phonetic = parts[1] or "",
          translation = parts[2] or "",
          exchange = parts[3] or "",
          detail = parts[4] or "",
        }
      end

      -- 获取有道在线例句
      local function fetch_youdao_examples(word)
        local encoded_word = word:gsub(" ", "%%20"):gsub("'", "%%27")
        local cmd = string.format(
          "curl -s -m 4 'https://dict.youdao.com/jsonapi?q=%s' -H 'User-Agent: Mozilla/5.0' 2>/dev/null",
          encoded_word
        )
        
        local output = vim.fn.system(cmd)
        if vim.v.shell_error ~= 0 or output == "" then return nil end
        
        local ok, json = pcall(vim.json.decode, output)
        if not ok or type(json) ~= "table" then return nil end
        
        local examples = {}
        if json.blng_sents_part and json.blng_sents_part["sentence-pair"] then
          for _, pair in ipairs(json.blng_sents_part["sentence-pair"]) do
            if #examples >= 3 then break end
            table.insert(examples, {
              en = pair.sentence,
              cn = pair["sentence-translation"],
            })
          end
        end
        
        return #examples > 0 and examples or nil
      end

      -- 修改：支持选择英音/美音
      -- accent: "uk" (英音, type=2) 或 "us" (美音, type=1)
      local function play_word_audio(word, accent)
        local suffix = accent == "us" and "us" or "uk"
        local audio_type = accent == "us" and "1" or "2"  -- 1=美音, 2=英音
        local accent_label = accent == "us" and "美音" or "英音"
        
        local audio_file = string.format("%s/%s_%s.mp3", audio_cache_dir, word:gsub("%s+", "_"), suffix)
        
        -- 下载音频
        if vim.fn.filereadable(audio_file) == 0 then
          local encoded_word = word:gsub(" ", "%%20")
          local url = string.format("https://dict.youdao.com/dictvoice?audio=%s&type=%s", encoded_word, audio_type)
          local cmd = string.format("curl -s -L -o '%s' '%s' 2>/dev/null", audio_file, url)
          
          vim.fn.system(cmd)
          
          if vim.v.shell_error ~= 0 or vim.fn.filereadable(audio_file) == 0 then
            vim.notify("下载" .. accent_label .. "失败", vim.log.levels.WARN)
            return
          end
        end
        
        -- 播放音频
        local play_cmd = nil
        if vim.fn.executable("mpv") == 1 then
          play_cmd = string.format("mpv --no-video --really-quiet '%s' &", audio_file)
        elseif vim.fn.executable("ffplay") == 1 then
          play_cmd = string.format("ffplay -nodisp -autoexit '%s' 2>/dev/null &", audio_file)
        elseif vim.fn.executable("paplay") == 1 then
          play_cmd = string.format("paplay '%s' &", audio_file)
        elseif vim.fn.executable("afplay") == 1 then
          play_cmd = string.format("afplay '%s' &", audio_file)
        elseif vim.fn.executable("cvlc") == 1 then
          play_cmd = string.format("cvlc --play-and-exit '%s' 2>/dev/null &", audio_file)
        end
        
        if play_cmd then
          vim.fn.system(play_cmd)
          vim.notify("🔊 " .. accent_label .. ": " .. word, vim.log.levels.INFO, { timeout = 800 })
        else
          vim.notify("未找到音频播放器（请安装 mpv）", vim.log.levels.WARN)
        end
      end

      local function show_result(word, data, youdao_examples)
        local lines = {}
        
        table.insert(lines, "----------------------------------------")
        table.insert(lines, "  " .. word)
        table.insert(lines, "----------------------------------------")
        
        if data.phonetic and data.phonetic ~= "" then
          table.insert(lines, "  音标: " .. data.phonetic)
        end
        
        if data.translation and data.translation ~= "" then
          for line in data.translation:gmatch("[^\n]+") do
            table.insert(lines, "  " .. line)
          end
        end
        
        if data.exchange and data.exchange ~= "" then
          table.insert(lines, "----------------------------------------")
          table.insert(lines, "  【词形变化】")
          for part in data.exchange:gmatch("[^/]+") do
            local key, val = part:match("(.+):(.+)")
            if key and val then
              local labels = {
                p = "过去式", d = "过去分词", i = "现在分词",
                ["3"] = "第三人称单数", s = "复数",
                r = "比较级", t = "最高级",
              }
              table.insert(lines, string.format("    %s: %s", labels[key] or key, val))
            end
          end
        end
        
        local has_local = data.detail and data.detail ~= ""
        local has_youdao = youdao_examples and #youdao_examples > 0
        
        if has_youdao or has_local then
          table.insert(lines, "----------------------------------------")
          table.insert(lines, "  【例句】")
          
          if has_youdao then
            for i, ex in ipairs(youdao_examples) do
              if #lines > 25 then break end
              table.insert(lines, string.format("  [%d] %s", i, ex.en))
              table.insert(lines, "      " .. ex.cn)
              table.insert(lines, "")
            end
          end
          
          if has_local then
            local ok, detail_json = pcall(vim.json.decode, data.detail)
            if ok and type(detail_json) == "table" then
              for pos, defs in pairs(detail_json) do
                for _, def in ipairs(defs) do
                  if #lines > 25 then break end
                  if def.example then
                    table.insert(lines, "  • " .. def.example)
                  end
                end
              end
            end
          end
        else
          table.insert(lines, "----------------------------------------")
          table.insert(lines, "  [无例句]")
        end
        
        table.insert(lines, "----------------------------------------")
        -- 修改：提示有两个发音选项
        table.insert(lines, "  [1]美音 [2]英音 [q]关闭")

        local buf = vim.api.nvim_create_buf(false, true)
        vim.api.nvim_buf_set_lines(buf, 0, -1, false, lines)
        vim.bo[buf].modifiable = false
        vim.bo[buf].filetype = "dictionary"
        
        vim.b[buf].dict_word = word
        
        local width = math.min(60, vim.o.columns - 4)
        local height = math.min(#lines, vim.o.lines - 4)
        
        local win = vim.api.nvim_open_win(buf, true, {
          relative = "editor",
          width = width,
          height = height,
          col = (vim.o.columns - width) / 2,
          row = (vim.o.lines - height) / 2,
          style = "minimal",
          border = "rounded",
          title = " 离线词典 ",
          title_pos = "center",
        })
        
        -- 按键映射
        vim.keymap.set("n", "q", "<cmd>close<cr>", { buffer = buf, silent = true })
        vim.keymap.set("n", "<Esc>", "<cmd>close<cr>", { buffer = buf, silent = true })
        
        -- 1 键：美音（type=1）
        vim.keymap.set("n", "1", function()
          local w = vim.b[buf].dict_word
          if w then play_word_audio(w, "us") end
        end, { buffer = buf, silent = true, desc = "播放美音" })
        
        -- 2 键：英音（type=2）
        vim.keymap.set("n", "2", function()
          local w = vim.b[buf].dict_word
          if w then play_word_audio(w, "uk") end
        end, { buffer = buf, silent = true, desc = "播放英音" })
      end

      local function translate(mode, input_word)
        local word
        if input_word then
          word = input_word
        elseif mode == "v" then
          local old_reg = vim.fn.getreg('"')
          vim.cmd('normal! ""y')
          word = vim.fn.getreg('"'):gsub("^%s*(.-)%s*$", "%1")
          vim.fn.setreg('"', old_reg)
        else
          word = vim.fn.expand("<cword>")
        end
        
        if not word or word == "" then return end
        
        vim.notify("查询: " .. word, vim.log.levels.INFO, { timeout = 500 })
        
        vim.defer_fn(function()
          local data, err = query_ecdict(word:lower())
          if not data then
            vim.notify(err or "查询失败", vim.log.levels.WARN)
            return
          end
          
          local youdao_examples = nil
          if not data.detail or data.detail == "" then
            vim.schedule(function()
              vim.notify("正在获取在线例句...", vim.log.levels.INFO, { timeout = 800 })
            end)
            youdao_examples = fetch_youdao_examples(word)
          end
          
          vim.schedule(function()
            show_result(word, data, youdao_examples)
          end)
        end, 10)
      end

      vim.keymap.set("n", "mm", function() translate("n") end, { desc = "词典查单词" })
      vim.keymap.set("v", "mm", function() translate("v") end, { desc = "词典查选中文本" })
      vim.keymap.set("n", "mi", function()
        vim.ui.input({ prompt = "输入单词: " }, function(input)
          if input then translate("n", input) end
        end)
      end, { desc = "词典输入查询" })
    end,
  },
}
